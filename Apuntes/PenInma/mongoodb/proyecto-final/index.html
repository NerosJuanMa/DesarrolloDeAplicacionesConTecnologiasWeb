📝 Plantillas de Código

🍃 Conexión MongoDB

// conexion-mongodb.js const { MongoClient } = require('mongodb'); const uri = 'mongodb://localhost:27017'; const client = new MongoClient(uri); async function connectToMongo() { try { await client.connect(); console.log('✅ Conectado a MongoDB'); return client.db('musicaDB'); } catch (error) { console.error('❌ Error conectando a MongoDB:', error); } } module.exports = { connectToMongo };

🗄️ Conexión MySQL

// conexion-mysql.js const mysql = require('mysql2/promise'); const config = { host: 'localhost', user: 'root', password: 'tu_password', database: 'musicaDB' }; async function connectToMySQL() { try { const connection = await mysql.createConnection(config); console.log('✅ Conectado a MySQL'); return connection; } catch (error) { console.error('❌ Error conectando a MySQL:', error); } } module.exports = { connectToMySQL };

🌐 API Express Básica

// server.js const express = require('express'); const cors = require('cors'); const { connectToMongo } = require('./conexion-mongodb'); const app = express(); const PORT = process.env.PORT || 3000; // Middlewares app.use(cors()); app.use(express.json()); // Rutas app.get('/api/canciones', async (req, res) => { try { const db = await connectToMongo(); const canciones = await db.collection('canciones').find().toArray(); res.json(canciones); } catch (error) { res.status(500).json({ error: error.message }); } }); app.listen(PORT, () => { console.log(`🚀 Servidor corriendo en puerto ${PORT}`); });
